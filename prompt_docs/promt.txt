# SECTION 5: SECURITY RULES SPECIFICATION

Agent, deploy the following production-ready security rules to your new Firebase project:

### 5.1 Cloud Firestore Security Rules (`firestore.rules`)
```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }
    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }
    function getUserData() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    function isPoliceOrAdmin() {
      return isAuthenticated() && (getUserData().role == 'police' || getUserData().role == 'admin');
    }
    function isAdmin() {
      return isAuthenticated() && getUserData().role == 'admin';
    }

    // Users Collection
    match /users/{userId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated();
      allow update: if isOwner(userId) || isAdmin();
      allow delete: if isAdmin();
    }

    // Complaints (CSR) Collection
    match /complaints/{complaintId} {
      allow read: if isAuthenticated() && (resource.data.citizenId == request.auth.uid || isPoliceOrAdmin() || resource.data.isBroadcasted == true);
      allow create: if isAuthenticated();
      allow update, delete: if isPoliceOrAdmin();
    }

    // FIR Incidents Collection
    match /incidents/{incidentId} {
      allow read: if isAuthenticated() && (resource.data.complainantId == request.auth.uid || isPoliceOrAdmin());
      allow create: if isAuthenticated();
      allow update, delete: if isPoliceOrAdmin();
    }

    // FIR Supporting Sub-collections & Tables
    match /offences/{id} { allow read, write: if isAuthenticated(); }
    match /evidence/{id} { allow read, write: if isAuthenticated(); }
    match /witnesses/{id} { allow read, write: if isAuthenticated(); }
    match /statements/{id} { allow read, write: if isAuthenticated(); }
    match /officer_reviews/{id} { 
      allow read: if isAuthenticated();
      allow write: if isPoliceOrAdmin();
    }

    // SOS Alerts Collection
    match /sos_alerts/{alertId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated();
      allow update: if isPoliceOrAdmin();
    }

    // Case Logs
    match /case_logs/{logId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isPoliceOrAdmin();
    }

    // Taxonomies
    match /categories/{id} { allow read: if true; allow write: if isAdmin(); }
    match /departments/{id} { allow read: if true; allow write: if isAdmin(); }
    match /areas/{id} { allow read: if true; allow write: if isAdmin(); }
  }
}
```

### 5.2 Firebase Storage Rules (`storage.rules`)
```javascript
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /evidence/{userId}/{allPaths=**} {
      allow read: if request.auth != null;

    }

      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }
    match /profile_documents/{userId}/{allPaths=**} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
  }
}
```

Follow this exact guide to connect a fresh Firebase project:

### Step 1: Create a New Firebase Project
1. Open the [Firebase Console](https://console.firebase.google.com/) and click **Add Project**.

### Step 2: Enable Firebase Services

3. **Cloud Storage:** Go to *Build > Storage > Get Started*. Accept default bucket settings.

### Step 3: Register Web App & Get Client Keys
1. In Project Settings (gear icon) > General, scroll to **Your apps** and click the Web icon (`</>`).
